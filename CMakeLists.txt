cmake_minimum_required(VERSION 3.20)
project(retro_study VERSION 0.3.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
include(cmake/ClapPackaging.cmake)

option(RETRO_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
  if(RETRO_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()

add_library(retro_dsp_objects OBJECT src/dsp/bitcolor.c src/dsp/curves.c src/dsp/noisegen.c src/dsp/relations.c src/dsp/digital_flux.c src/dsp/wobble.c src/dsp/wobble_macro.c)
target_include_directories(retro_dsp_objects PUBLIC src/dsp)
add_library(retro_dsp SHARED $<TARGET_OBJECTS:retro_dsp_objects>)
target_include_directories(retro_dsp PUBLIC src/dsp)
target_link_libraries(retro_dsp PRIVATE m)

add_library(retro_digital MODULE src/clap/plugin.c src/dsp/bitcolor.c src/dsp/curves.c src/dsp/relations.c src/dsp/noisegen.c src/dsp/digital_flux.c)
target_include_directories(retro_digital PRIVATE src/dsp third_party/clap/include)
target_link_libraries(retro_digital PRIVATE m)
set_target_properties(retro_digital PROPERTIES RETRO_CLAP_VERSION 0.3.0)
retro_configure_clap(retro_digital RetroStudyDigital local.retro-study.digital)

add_library(retro_wobble MODULE src/clap/wobble_plugin.c src/dsp/wobble.c src/dsp/curves.c src/dsp/wobble_macro.c src/dsp/noisegen.c src/dsp/relations.c)
target_include_directories(retro_wobble PRIVATE src/dsp third_party/clap/include)
target_link_libraries(retro_wobble PRIVATE m)
set_target_properties(retro_wobble PROPERTIES RETRO_CLAP_VERSION 0.2.0)
retro_configure_clap(retro_wobble RetroStudyWobble local.retro-study.wobble)

if(BUILD_TESTING)
  add_executable(clap_host_test tests/clap_host.c)
  target_include_directories(clap_host_test PRIVATE third_party/clap/include src/dsp)
  target_link_libraries(clap_host_test PRIVATE retro_dsp ${CMAKE_DL_LIBS} m)
  # assert() is the test runner's check mechanism, including in Release builds.
  target_compile_options(clap_host_test PRIVATE -UNDEBUG)
  add_test(NAME clap_host COMMAND clap_host_test $<TARGET_FILE:retro_digital>)
  add_executable(dsp_test tests/dsp_test.c)
  target_link_libraries(dsp_test PRIVATE retro_dsp m)
  target_compile_options(dsp_test PRIVATE -UNDEBUG)
  add_test(NAME dsp_contract COMMAND dsp_test)
  add_executable(wobble_host_test tests/wobble_host.c)
  target_include_directories(wobble_host_test PRIVATE third_party/clap/include src/dsp)
  target_link_libraries(wobble_host_test PRIVATE retro_dsp ${CMAKE_DL_LIBS} m)
  target_compile_options(wobble_host_test PRIVATE -UNDEBUG)
  add_test(NAME wobble_host COMMAND wobble_host_test $<TARGET_FILE:retro_wobble>)
endif()
